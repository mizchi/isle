不正入力・必須同意・送信成功を、ラベル付きのフォームで試す。 app.isle と vite.config.mjs を一緒に読む。状態・表示の宣言はコンパイルされ、名前付き handler が追跡できるコード境界になる。

リポジトリ内の例は、ルートで上記コマンドを実行する。検査は just a11y-form。利用者のアプリへ組み込む場合は npm パッケージの導入手順を使い、このディレクトリのソースと設定を参照する。

実装を読む

app.isle
export client component Signup({ initial }: { initial: string }) {
  state email = initial;
  state accepted = false;
  state invalid = false;
  state error = "";
  state submitted = "";
  <form aria-label="Signup">
    on submit save { submitted = email; }
    <label for="email">Email address (required)</label>
    <input id="email" name="email" type="email" autocomplete="email" required
      minlength="6" maxlength="80" value={email} aria-invalid={invalid}
      aria-describedby="email-help email-error">
      on input edit { email = event.value; invalid = event.invalid; error = event.validationMessage; }
      on invalid report { invalid = event.invalid; error = event.validationMessage; }
    </input>
    <p id="email-help">Use your work email.</p>
    <p id="email-error" aria-live="polite">{error}</p>
    <label>
      <input name="accepted" type="checkbox" checked={accepted} required>
        on change consent { accepted = event.checked; }
      </input>
      I agree (required)
    </label>
    <button type="submit">Save</button>
    <output aria-live="polite">{submitted}</output>
  </form>
}
main.mjs
import { boot } from 'virtual:isle-loader';
window.isleDemo = boot();
vite.config.mjs
import { defineConfig } from 'vite';
import { fileURLToPath } from 'node:url';
import { isle } from '@isle/prototype/vite';

export default defineConfig({
  root: fileURLToPath(new URL('.', import.meta.url)),
  plugins: [isle({ entry: fileURLToPath(new URL('./app.isle', import.meta.url)),
    diagnostics: true, clientEntry: fileURLToPath(new URL('./main.mjs', import.meta.url)),
    mounts: [{ component: 'Signup', props: { initial: '' }, scopeId: 'signup', label: 'Signup' }] })],
  server: { host: '127.0.0.1', port: 4190, strictPort: true,
    fs: { allow: [fileURLToPath(new URL('../..', import.meta.url))] } },
  build: { outDir: '../../dist-form', emptyOutDir: true, manifest: true },
});
