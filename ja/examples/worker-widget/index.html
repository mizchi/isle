Workers からの widget 配信

widget-dev と widget-host を別 terminal で起動し、別 origin に固定サイズの Shadow DOM widget を埋め込む。

この例にはサーバーが必要です。GitHub Pages には RPC endpoint を置かず、下の host と実行手順を使います。

widget-dev と widget-host を別 terminal で起動し、別 origin に固定サイズの Shadow DOM widget を埋め込む。 別 terminal で just widget-host を起動して http://127.0.0.1:8788/ を開く。widget 自体は8787で動く。width/height で領域を予約し、runtime の版は完全一致させる。host policy は配備ガイドを参照。

リポジトリ内の例は、ルートで上記コマンドを実行する。検査は just widget-check。利用者のアプリへ組み込む場合は npm パッケージの導入手順を使い、このディレクトリのソースと設定を参照する。

This larger example uses the dashboard source and dataset.

大きい widget の例なので、上記の dashboard ソースと dataset も参照する。

import { mkdir, writeFile } from 'node:fs/promises';

await build({ ...config(), configFile: false });

import { renderWidgetEmbed } from '@isle/prototype/widget';

export function renderHostPage(widgetURL, { width = 640, height = 640, loading = 'eager', count = 1, runtime = 'shared' } = {}) {
  const widgets = Array.from({ length: count }, (_, i) => renderWidgetEmbed({ src: widgetURL, width, height, loading, runtime, title: `Work planner ${i + 1}` })).join('\n');
  return `<!doctype html><html lang="en"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><link rel="icon" href="data:,"><title>Widget publisher demo</title><style>
  *{box-sizing:border-box}body{font:16px/1.6 system-ui,sans-serif;color:#182b3b;background:#edf2f7;margin:0}main{max-width:900px;margin:auto;padding:24px 16px}h1{font-size:28px;line-height:1.3}isle-widget,isle-widget-standalone{background:white;margin:24px 0}#after{padding:16px;border:1px solid #bccbd8;border-radius:8px;background:white}a{color:#185891}
  </style></head><body><main><h1>Widget publisher demo</h1><p>This page belongs to the publisher. The work planner below is a Web Component served by Cloudflare Workers.</p>
  ${widgets}<p id="after">Following content keeps its place while the widget loads and changes.</p></main></body></html>`;
}
host.mjs
/** Public stateless demo. Real authentication/persistence belongs in a separate host adapter. */
export function createDemoHost(ids) {
  const tasks = new Set(ids);
  return {
    context: () => ({ subject: 'public-demo', tenant: 'widget-demo' }),
    policies: {
      'policy.planEditor': (context, id) => context.tenant === 'widget-demo' && tasks.has(id),
      'policy.projectCreator': context => context.tenant === 'widget-demo',
    },
    validators: { 'projects.available': { effects: ['db.read'], validate: ({ input }) => input.project === 'Taken project' ? [{ field: 'project', code: 'taken' }] : [] } },
    operations: {
      'plans.save': { policy: 'policy.planEditor', effects: ['db.write'], commit: ({ args }) => ({ hours: args[1] }) },
      'projects.create': { policy: 'policy.projectCreator', effects: ['db.write'], commit: ({ args }) => ({ project: args[1] }) },
    },
  };
}
vite.config.mjs
import { defineConfig } from 'vite';
import { fileURLToPath } from 'node:url';
import { isle } from '@isle/prototype/vite';
import { tasks, teams } from '../../benchmarks/dashboard/model.mjs';
import { styles, dashboardStyles } from '../../benchmarks/dashboard/shell.mjs';

const root = fileURLToPath(new URL('.', import.meta.url));
const markup = `<main><h1>Work planner widget</h1>
<p class="intro">Public demo. Changes reset when this widget reloads.</p>
<fieldset><legend>Filter work packages by team</legend><label><input type="radio" id="team-all" name="team-filter" checked>All teams</label>${teams.map((team, i) => `<label><input type="radio" id="team-${i}" name="team-filter">${team}</label>`).join('')}</fieldset>
<!--isle:ssr--><footer>64 independent work packages. No sign-in or third-party cookies required.</footer></main>`;
export function config(runtime = 'shared') { return defineConfig({ root, base: './', plugins: [
  { name: 'widget-shell', transformIndexHtml: { order: 'pre', handler: html => html.replace('</head>', `<style>${styles}</style></head>`).replace('<!--widget:shell-->', markup) } },
  isle({ entry: fileURLToPath(new URL('../../benchmarks/variants/isle/dashboard.isle', import.meta.url)), loading: 'eager',
    widget: {
      runtime,
      actionBaseURL: runtime === 'standalone' ? '../__isle/' : './__isle/',
      template: `<div class="widget-root">${markup}</div>`,
      css: ':host{container-type:inline-size;color-scheme:light}.widget-root{font:16px system-ui,sans-serif;text-align:left}' + dashboardStyles({ root: '.widget-root', body: '.widget-root', compact: '@container (max-width:600px)' }),
    }, mounts: [
    { component: 'ProjectForm', props: {}, scopeId: 'project', label: 'Create a project' },
    ...tasks.map(task => ({ component: 'WorkItem', props: task, scopeId: task.id, label: task.title })),
  ] }),
], build: { outDir: `../../dist/worker-widget/client${runtime === 'standalone' ? '/standalone' : ''}`, emptyOutDir: true, target: 'es2023', sourcemap: false } }); }
export default config();

import contracts from '../../dist/worker-widget/server/contracts.json' with { type: 'json' };
