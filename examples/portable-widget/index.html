Portable Web Components

Source / ソース: server.mjs · static.isle · vite.config.mjs

Mount independent widgets, compare shared and standalone distributions, and reject a version mismatch. Open http://127.0.0.1:8793/portable/. Its Node server serves a custom asset path, shared/standalone builds and a mismatch case. The descriptor and generated loader are the public embed interface.

From a repository checkout, run at the repository root:

sh
just widget-portable-dev

Open http://127.0.0.1:8793/portable/.

Check: just widget-check. For a consumer app, install the package and use the npm setup; the source files here show the feature and its configuration.

server.mjs
// Node adapter example. Core RPC depends only on standard Request/Response.
import { createServer } from 'node:http';
import { readFile } from 'node:fs/promises';
import { resolve, sep } from 'node:path';
import { fileURLToPath } from 'node:url';
import contracts from '../../dist-portable-widget/server/contracts.json' with { type: 'json' };
import { createActionHandler } from '@isle/prototype/server';

const actionBasePath = '/portable/api/actions/';
const handle = createActionHandler(contracts, {
  context: () => ({ subject: 'public-demo', tenant: 'portable' }),
  policies: { 'policy.documentEditor': (context, id) => context.tenant === 'portable' && id === 'portable-demo' },
  operations: { 'documents.rename': { policy: 'policy.documentEditor', effects: ['db.write'], commit: ({ args }) => ({ id: 'portable-demo', title: args[1] }) } },
}, { allowedOrigins: '*', actionBasePath });
const client = resolve(fileURLToPath(new URL('../../dist-portable-widget/client/', import.meta.url)));
const server = createServer(async (request, response) => {
  const origin = `http://${request.headers.host}`, url = new URL(request.url, origin);
  if (url.pathname.startsWith(actionBasePath)) {
    const chunks = []; for await (const chunk of request) chunks.push(chunk);
    const wire = new Request(url, { method: request.method, headers: request.headers, ...(request.method === 'POST' ? { body: Buffer.concat(chunks) } : {}) });
    const result = await handle(wire);
    response.writeHead(result.status, Object.fromEntries(result.headers)); response.end(await result.text()); return;
  }
  try {
    if (!['GET', 'HEAD'].includes(request.method) || !url.pathname.startsWith('/portable/')) throw new Error('Not found');
    const path = decodeURIComponent(url.pathname.slice('/portable/'.length)) || 'index.html';
    const file = resolve(client, path);
    if (!file.startsWith(client + sep)) throw new Error('Not found');
    const body = await readFile(file);
    const type = path.endsWith('.js') ? 'text/javascript' : path.endsWith('.json') ? 'application/json' : 'text/html';
    response.writeHead(200, { 'content-type': `${type}; charset=utf-8`, 'access-control-allow-origin': '*', 'cache-control': 'no-cache' });
    response.end(request.method === 'HEAD' ? undefined : body);
  } catch { response.writeHead(404); response.end('Not found'); }
});
server.listen(Number(process.env.PORT ?? 8793), '127.0.0.1', () => console.log(`Portable widget: http://127.0.0.1:${server.address().port}/portable/`));
for (const signal of ['SIGINT', 'SIGTERM']) process.on(signal, () => server.close(() => process.exit()));

vite.config.mjs
import { fileURLToPath } from 'node:url';
import { isle } from '@isle/prototype/vite';
import { runtimeIdentity } from '../../runtime/runtime-identity.mjs';
export const alternateVersion = `${runtimeIdentity.version}-fixture`;

export function config(staticMode = false, { runtime = 'shared', runtimeVersion = runtimeIdentity.version, distribution = staticMode ? 'static' : '', tagName } = {}) {
  const root = fileURLToPath(new URL('.', import.meta.url));
  const title = staticMode ? 'Static notice' : 'Portable counter';
  const template = `<main><h1>${title}</h1><!--isle:ssr--></main>`;
  const css = ':host{color-scheme:light}*{box-sizing:border-box}body{margin:0}main{font:16px/1.6 system-ui,sans-serif;color:#182b3b;background:#f3f6fa;padding:24px}h1{font-size:24px;line-height:1.2;margin:0 0 20px}h2{font-size:18px;margin:16px 0 8px}button{font:inherit;min-height:44px;min-width:120px;padding:8px 16px;border:1px solid #224e72;border-radius:6px;background:#193f60;color:white}button:focus-visible{outline:3px solid #a44f00;outline-offset:3px}';
  return { root, base: './', plugins: [
    { name: 'portable-widget-shell', transformIndexHtml: { order: 'pre', handler: html => html.replace('</head>', `<style>${css}</style></head>`).replace('<!--widget:shell-->', template) } },
    isle({ entry: fileURLToPath(new URL(staticMode ? 'static.isle' : '../counter/app.isle', import.meta.url)),
      mode: staticMode ? 'ssr' : 'resume', loading: 'eager', clientEntry: staticMode ? undefined : '/page.mjs',
      mounts: staticMode ? [{ component: 'Notice', props: { message: 'This widget has no UI runtime or handler chunks.' }, label: 'Server output' }] : [
        { component: 'Counter', props: { initial: 17 }, scopeId: 'count', label: 'Counter' },
        { component: 'Editor', props: { id: 'portable-demo', initial: 'Draft' }, scopeId: 'edit', label: 'Editor' },
      ],
      widget: { runtime, runtimeVersion, tagName: tagName ?? `${staticMode ? 'static-counter' : 'demo-counter'}${distribution && distribution !== 'static' ? `-${distribution}` : ''}`, descriptorFile: `components/${staticMode ? 'static' : 'counter'}.json`,
        loaderFile: `sdk/${staticMode ? 'static' : 'counter'}.js`, actionBaseURL: distribution ? '../../api/actions/' : '../api/actions/', template, css },
    }),
  ], build: { outDir: `../../dist-portable-widget/client${distribution ? `/${distribution}` : ''}`, emptyOutDir: true, target: 'es2023' } };
}
export default config();
